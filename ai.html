# app.py import os import numpy as np from pypdf import PdfReader from openai import OpenAI client = OpenAI(api_key=os.environ["OPENAI_API_KEY"]) CHUNK_SIZE = 1000 TOP_K = 5 def load_text(pdf_path: str) -> str: reader = PdfReader(pdf_path) return "\n".join(page.extract_text() or "" for page in reader.pages) def chunk_text(text: str, size: int = CHUNK_SIZE) -> list[str]: words = text.split() return [" ".join(words[i:i + size]) for i in range(0, len(words), size)] def embed(texts: list[str]) -> np.ndarray: resp = client.embeddings.create(model="text-embedding-3-small", input=texts) return np.array([d.embedding for d in resp.data]) def build_index(pdf_path: str): chunks = chunk_text(load_text(pdf_path)) vectors = embed(chunks) vectors /= np.linalg.norm(vectors, axis=1, keepdims=True) return chunks, vectors def answer(question: str, chunks, vectors) -> str: q = embed([question])[0] q /= np.linalg.norm(q) top = np.argsort(vectors @ q)[::-1][:TOP_K] context = "\n\n---\n\n".join(chunks[i] for i in top) resp = client.chat.completions.create( model="gpt-4o-mini", messages=[ {"role": "system", "content": "You answer questions about the FIRST Tech Challenge Competition Manual. " "Use only the provided context. If the answer is not in the context, say so " "and suggest which manual section to check."}, {"role": "user", "content": f"Context:\n{context}\n\nQuestion: {question}"}, ], ) return resp.choices[0].message.content if __name__ == "__main__": chunks, vectors = build_index("Competition_Manual.pdf") print("Ask a question about the manual (type 'exit' to quit).") while True: q = input("> ").strip() if q.lower() == "exit": break print(answer(q, chunks, vectors), "\n")
