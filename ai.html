# app.py import os import numpy as np from flask import Flask, request, jsonify from pypdf import PdfReader from openai import OpenAI PDF_PATH = "Competition_Manual.pdf" CHUNK_WORDS = 1000 TOP_K = 5 EMBED_MODEL = "text-embedding-3-small" CHAT_MODEL = "gpt-4o-mini" client = OpenAI(api_key=os.environ["OPENAI_API_KEY"]) app = Flask(__name__) # ---------- Manual indexing ---------- def load_text(pdf_path: str) -> str: reader = PdfReader(pdf_path) return "\n".join(page.extract_text() or "" for page in reader.pages) def chunk_text(text: str, size: int = CHUNK_WORDS) -> list[str]: words = text.split() return [" ".join(words[i:i + size]) for i in range(0, len(words), size)] def embed(texts: list[str]) -> np.ndarray: resp = client.embeddings.create(model=EMBED_MODEL, input=texts) return np.array([d.embedding for d in resp.data]) def build_index(pdf_path: str = PDF_PATH): chunks = chunk_text(load_text(pdf_path)) vectors = embed(chunks) vectors = vectors / np.linalg.norm(vectors, axis=1, keepdims=True) return chunks, vectors # Build the index once when the server starts CHUNKS, VECTORS = build_index(PDF_PATH) # ---------- Answering ---------- def answer(question: str) -> str: q = embed([question])[0] q = q / np.linalg.norm(q) top = np.argsort(VECTORS @ q)[::-1][:TOP_K] context = "\n\n---\n\n".join(CHUNKS[i] for i in top) resp = client.chat.completions.create( model=CHAT_MODEL, messages=[ { "role": "system", "content": ( "You answer questions about the FIRST Tech Challenge Competition Manual. " "Use only the provided context. If the answer is not in the context, say so " "and suggest which manual section to check." ), }, {"role": "user", "content": f"Context:\n{context}\n\nQuestion: {question}"}, ], ) return resp.choices[0].message.content # ---------- Web page ---------- HTML_PAGE = """ FTC Manual Assistant

FTC Competition Manual Assistant

Ask a question about the 2026-2027 FIRST Tech Challenge Competition Manual.

""" @app.route("/") def home(): return HTML_PAGE @app.route("/ask", methods=["POST"]) def ask(): data = request.get_json(silent=True) or {} question = (data.get("question") or "").strip() if not question: return jsonify({"error": "Please enter a question."}), 400 try: return jsonify({"answer": answer(question)}) except Exception as e: return jsonify({"error": f"Something went wrong: {e}"}), 500 if __name__ == "__main__": app.run(debug=True, port=5000)
